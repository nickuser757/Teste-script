local mt = getrawmetatable(game)
local oldNamecall = mt.__namecall
setreadonly(mt, false)

mt.__namecall = newcclosure(function(self, ...)
    local method = getnamecallmethod()
    local args = {...}
    
    if tostring(method) == "Kick" or tostring(method) == "kick" then
        print("Tentativa de Kick bloqueada: " .. tostring(args[1]))
        return nil -- Ignora o comando de kick
    end
    
    return oldNamecall(self, ...)
end)

-- Lista simplificada e eficiente de alvos (focado em scripts e remotes)
local function CleanAntiCheat()
    local targets = {
        game:GetService("Players").LocalPlayer:WaitForChild("PlayerScripts"),
        game:GetService("ReplicatedStorage"),
        game:GetService("StarterGui")
    }

    for _, service in ipairs(targets) do
        if not ACRunning then break end
        
        for _, item in ipairs(service:GetDescendants()) do
            -- Procura por nomes comuns de anti-cheat
            local name = item.Name:lower()
            if name:find("anticheat") or name:find("exploit") or name:find("ac_") or name:find("catch") or name:find("adonis") then
                pcall(function()
                    item:Destroy() -- Deleta o script de segurança
                end)
            end
        end
        task.wait(0.1) -- Pausa curta entre serviços para o ping não subir
    end
end

-- Loop principal
task.spawn(function()
    while ACRunning do
        pcall(CleanAntiCheat)
        task.wait(ScanDelay) -- Espera antes de escanear de novo
    end
end)

local function AtivarFPSBoost()
    local Lighting = game:GetService("Lighting")
    local Terrain = workspace:FindFirstChildOfClass('Terrain')
    
    Terrain.WaterWaveSize = 0
    Terrain.WaterWaveSpeed = 0
    Terrain.WaterReflectance = 0
    Terrain.WaterTransparency = 0
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 9e9
    settings().Rendering.QualityLevel = Enum.QualityLevel.Level01

    for _, v in pairs(game:GetDescendants()) do
        if v:IsA("Part") or v:IsA("UnionOperation") or v:IsA("MeshPart") then
            v.Material = Enum.Material.SmoothPlastic
            v.Reflectance = 0
        elseif v:IsA("Decal") or v:IsA("Texture") then
            v.Transparency = 1
        elseif v:IsA("ParticleEmitter") or v:IsA("Trail") then
            v.Enabled = false
        end
    end
end

-- Função Ninja para detectar o time (Busca por nome ou cor do time)
local function IdentificarCorDoTime(player)
    local team = player.Team
    local teamName = team and team.Name:lower() or ""
    local teamColor = team and team.TeamColor.Name or ""

    -- Lógica para o HOMER (Amarelo)
    if string.find(teamName, "homer") or teamColor:find("Yellow") or teamColor:find("Gold") then
        return Cores.Homer
    -- Lógica para o BART (Vermelho)
    elseif string.find(teamName, "bart") or teamColor:find("Red") or teamColor:find("Crimson") then
        return Cores.Bart
    end
    
    return Cores.Fallback
end

local function AdicionarESP(player)
    if player == LocalPlayer then return end

    local function aplicar(character)
        if not character then return end
        local root = character:WaitForChild("HumanoidRootPart", 10)
        local head = character:WaitForChild("Head", 10)
        if not root or not head then return end

        local corDoTime = IdentificarCorDoTime(player)

        -- 1. HIGHLIGHT (O brilho em volta do corpo)
        local highlight = character:FindFirstChild("HardcoreHighlight") or Instance.new("Highlight")
        highlight.Name = "HardcoreHighlight"
        highlight.Adornee = character
        highlight.FillColor = corDoTime
        highlight.OutlineColor = Color3.new(1, 1, 1)
        highlight.FillTransparency = 0.4
        highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        highlight.Parent = character

        -- 2. BILLBOARD GUI (O Nome flutuante)
        local billboard = head:FindFirstChild("NameESP") or Instance.new("BillboardGui")
        billboard.Name = "NameESP"
        billboard.Adornee = head
        billboard.Size = UDim2.new(0, 150, 0, 50)
        billboard.StudsOffset = Vector3.new(0, 3, 0)
        billboard.AlwaysOnTop = true
        billboard.Parent = head

        local label = billboard:FindFirstChild("TextLabel") or Instance.new("TextLabel")
        label.Parent = billboard
        label.BackgroundTransparency = 1
        label.Size = UDim2.new(1, 0, 1, 0)
        label.Text = (player.DisplayName or player.Name)
        label.TextColor3 = corDoTime -- Cor do texto igual a do time
        label.TextStrokeTransparency = 0
        label.TextSize = 16
        label.Font = Enum.Font.GothamBold
    end

    aplicar(player.Character)
    player.CharacterAdded:Connect(aplicar)
end

-- FUNÇÕES DE CONTROLE (CHAME PELO SEU TOGGLE)

function AtivarESP()
    DesativarESP() -- Limpeza preventiva
    
    for _, player in pairs(Players:GetPlayers()) do
        AdicionarESP(player)
    end

    table.insert(Connections, Players.PlayerAdded:Connect(AdicionarESP))
    print("✅ ESP COLORIDO ATIVADO: AMARELO (HOMER) | VERMELHO (BART)")
end

function DesativarESP()
    for _, conn in pairs(Connections) do if conn then conn:Disconnect() end end
    Connections = {}

    for _, player in pairs(Players:GetPlayers()) do
        local char = player.Character
        if char then
            local h = char:FindFirstChild("HardcoreHighlight")
            if h then h:Destroy() end
            if char:FindFirstChild("Head") then
                local n = char.Head:FindFirstChild("NameESP")
                if n then n:Destroy() end
            end
        end
    end
    print("❌ ESP DESATIVADO")
end

local FullBrightEnabled = false

function AtivarFullBright()
    FullBrightEnabled = true
    
    local function apply()
        if not FullBrightEnabled then return end
        Light.Ambient = Color3.new(1, 1, 1)
        Light.ColorShift_Bottom = Color3.new(1, 1, 1)
        Light.ColorShift_Top = Color3.new(1, 1, 1)
        Light.Brightness = 3 -- Brilho aumentado para ficar bem claro
        Light.OutdoorAmbient = Color3.new(1, 1, 1) -- Remove as sombras de fora
    end

    apply()
    -- Conecta para o caso do jogo tentar mudar o tempo/iluminação
    LightConnection = Light.Changed:Connect(apply)
    print("☀️ FULLBRIGHT: ATIVADO")
end

function DesativarFullBright()
    FullBrightEnabled = false
    if LightConnection then 
        LightConnection:Disconnect() 
        LightConnection = nil
    end
    
    -- Restaura a iluminação original do jogo
    Light.Ambient = OriginalSettings.Ambient
    Light.ColorShift_Bottom = OriginalSettings.ColorShift_Bottom
    Light.ColorShift_Top = OriginalSettings.ColorShift_Top
    Light.Brightness = OriginalSettings.Brightness
    Light.OutdoorAmbient = OriginalSettings.OutdoorAmbient
    print("🌙 FULLBRIGHT: DESATIVADO")
end
