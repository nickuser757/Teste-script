-- ESP Helper Functions
local function createESP(character, teamName)
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    
    local espKey = character.Name .. "_" .. teamName
    
    -- Remove existing ESP if it exists
    if ESPObjects[espKey] then
        for _, obj in pairs(ESPObjects[espKey]) do
            if obj and obj.Parent then
                obj:Destroy()
            end
        end
        ESPObjects[espKey] = nil
    end
    
    local espTable = {}
    
    -- Create Highlight
    local highlight = Instance.new("Highlight")
    highlight.Name = "ESP_Highlight"
    highlight.Adornee = character
    highlight.FillTransparency = 0.5
    highlight.OutlineTransparency = 0
    
    if teamName == "Homer" then
        highlight.FillColor = Color3.fromRGB(255, 0, 0)
        highlight.OutlineColor = Color3.fromRGB(255, 0, 0)
    elseif teamName == "Bart" then
        highlight.FillColor = Color3.fromRGB(255, 255, 255)
        highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    end
    
    highlight.Parent = character
    table.insert(espTable, highlight)
    
    -- Create BillboardGui for text label
    local billboard = Instance.new("BillboardGui")
    billboard.Name = "ESP_Billboard"
    billboard.Adornee = hrp
    billboard.Size = UDim2.new(0, 100, 0, 50)
    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.AlwaysOnTop = true
    billboard.Parent = hrp
    
    -- Create TextLabel
    local textLabel = Instance.new("TextLabel")
    textLabel.Name = "ESP_Label"
    textLabel.Size = UDim2.new(1, 0, 1, 0)
    textLabel.BackgroundTransparency = 1
    textLabel.Font = Enum.Font.GothamBold
    textLabel.TextSize = 18
    textLabel.Text = teamName
    
    if teamName == "Homer" then
        textLabel.TextColor3 = Color3.fromRGB(255, 0, 0)
    elseif teamName == "Bart" then
        textLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    end
    
    textLabel.TextStrokeTransparency = 0
    textLabel.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    textLabel.Parent = billboard
    
    table.insert(espTable, billboard)
    
    ESPObjects[espKey] = espTable
end

local function removeESP(character, teamName)
    local espKey = character.Name .. "_" .. teamName
    if ESPObjects[espKey] then
        for _, obj in pairs(ESPObjects[espKey]) do
            if obj and obj.Parent then
                obj:Destroy()
            end
        end
        ESPObjects[espKey] = nil
    end
end

local function removeAllESP()
    for key, espTable in pairs(ESPObjects) do
        for _, obj in pairs(espTable) do
            if obj and obj.Parent then
                obj:Destroy()
            end
        end
    end
    ESPObjects = {}
end

local function updateESP()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local character = player.Character
            local humanoid = character:FindFirstChild("Humanoid")
            
            if humanoid and humanoid.Health > 0 then
                if player.Team == HomerTeam and HomerESPEnabled then
                    createESP(character, "Homer")
                elseif player.Team == BartTeam and BartESPEnabled then
                    createESP(character, "Bart")
                else
                    -- Remove ESP if player is on Dead team or ESP is disabled
                    removeESP(character, "Homer")
                    removeESP(character, "Bart")
                end
            else
                -- Remove ESP if player is dead
                removeESP(character, "Homer")
                removeESP(character, "Bart")
            end
        end
    end
end

                    return
                end
                
                local Character = LocalPlayer.Character
                if Character then
                    for _, part in pairs(Character:GetDescendants()) do
                        if part:IsA('BasePart') then
                            part.CanCollide = false
                        end
                    end
                end
            end)
        else
            -- Disconnect the noclip connection
            if NoclipConnection then
                NoclipConnection:Disconnect()
                NoclipConnection = nil
            end
            
            -- Re-enable collision for all parts
            local Character = LocalPlayer.Character
            if Character then
                for _, part in pairs(Character:GetDescendants()) do
                    if part:IsA('BasePart') then
                        -- Only re-enable collision for parts that should have it
                        if part.Name == "Head" or part.Name:find("Arm") or part.Name:find("Leg") or part.Name:find("Torso") then
                            part.CanCollide = true
                        end
                        -- HumanoidRootPart should stay CanCollide = false
                        if part.Name == "HumanoidRootPart" then
                            part.CanCollide = false
                        end
                    end
                end
            end
        end
    end,
})

                    -- Find any Bart player to teleport to

                            local char = player.Character
                            if char then
                                local hrp = char:FindFirstChild("HumanoidRootPart")
                                local humanoid = char:FindFirstChild("Humanoid")
                                
                                if hrp and humanoid and humanoid.Health > 0 then
                                    -- Teleport directly to the Bart player
                                    myHRP.CFrame = hrp.CFrame

                            end
                        end
                    end
                end
            end)
        end

                                local humanoid = char:FindFirstChild("Humanoid")

                                    local killRemote = ReplicatedStorage:FindFirstChild("KillPlayer") or

local RitualSpeed = 0.01
local RitualEnabled = false
local RitualNoclipConnection = nil

AutomationTab:Slider({
    Title = 'Ritual Speed',
    Step = 0.001,
    Value = {
        Min = 0.001,
        Default = 0.01,
        Max = 0.1,
    },
    Callback = function(value)
        RitualSpeed = value
    end,
})

AutomationTab:Toggle({
    Title = 'Ritual (Homer)',
    Default = false,
    Callback = function(enabled)
        RitualEnabled = enabled
        
        if enabled then
            -- Enable noclip for ritual
            if LocalPlayer.Character then
                for _, part in ipairs(LocalPlayer.Character:GetDescendants()) do
                    if part:IsA('BasePart') then
                        part.CanCollide = false
                    end
                end
            end
            
            -- Disconnect old connection if exists
            if RitualNoclipConnection then
                RitualNoclipConnection:Disconnect()
            end
            
            -- Keep noclip active
            RitualNoclipConnection = RunService.Stepped:Connect(function()
                if not RitualEnabled then
                    return
                end
                
                if LocalPlayer.Character then
                    for _, part in ipairs(LocalPlayer.Character:GetDescendants()) do
                        if part:IsA('BasePart') then
                            part.CanCollide = false
                        end
                    end
                end
            end)
            
            -- Handle character respawn
            LocalPlayer.CharacterAdded:Connect(function(character)
                if RitualEnabled then
                    for _, part in ipairs(character:GetDescendants()) do
                        if part:IsA('BasePart') then
                            part.CanCollide = false
                        end
                    end
                end
            end)
            
            -- Main ritual loop
            task.spawn(function()
                local angle = 0
                local ritualRadius = 17 -- 17 studs from Homer as per tooltip
                
                while RitualEnabled do
                    task.wait(RitualSpeed)
                    
                    if not LocalPlayer.Character then continue end
                    if LocalPlayer.Team and LocalPlayer.Team.Name ~= "Bart" then continue end
                    
                    local myHRP = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                    if not myHRP then continue end
                    
                    -- Find closest Homer player
                    local closestHomer = nil
                    local closestDistance = math.huge
                    
                    for _, player in ipairs(Players:GetPlayers()) do
                        if player.Team and player.Team.Name == "Homer" then
                            local char = player.Character
                            if char then
                                local hrp = char:FindFirstChild("HumanoidRootPart")
                                if hrp then
                                    local distance = (myHRP.Position - hrp.Position).Magnitude
                                    if distance < closestDistance then
                                        closestDistance = distance
                                        closestHomer = hrp
                                    end
                                end
                            end
                        end
                    end
                    
                    -- Circle around the Homer
                    if closestHomer then
                        local x = math.cos(angle) * ritualRadius
                        local z = math.sin(angle) * ritualRadius
                        
                        local targetPos = closestHomer.Position + Vector3.new(x, 0, z)
                        myHRP.CFrame = CFrame.new(targetPos, closestHomer.Position)
                        
                        angle = angle + 0.1
                        if angle >= math.pi * 2 then
                            angle = 0
                        end
                    end
                end
            end)
        else
            -- Disconnect the ritual noclip connection
            if RitualNoclipConnection then
                RitualNoclipConnection:Disconnect()
                RitualNoclipConnection = nil
            end
            
            -- Re-enable collision when disabled
            if LocalPlayer.Character then
                for _, part in ipairs(LocalPlayer.Character:GetDescendants()) do
                    if part:IsA('BasePart') then
                        -- Only re-enable collision for parts that should have it
                        if part.Name == "Head" or part.Name:find("Arm") or part.Name:find("Leg") or part.Name:find("Torso") then
                            part.CanCollide = true
                        end
                        -- HumanoidRootPart should stay CanCollide = false
                        if part.Name == "HumanoidRootPart" then
                            part.CanCollide = false
                        end
                    end
                end
            end
        end
    end,
})
